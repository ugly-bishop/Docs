Executes an outbound bank transfer or international payout. Requires merchant approval in live environments.

### Scopes required

`payouts:write` or `*`

### Headers

<ParamField header="Authorization" type="string" required>
  Bearer API secret key formatted as `Bearer sk_live_...` or `Bearer sk_test_...`.
</ParamField>
<ParamField header="Idempotency-Key" type="string">
  Unique UUID v4 string to prevent duplicate transfers during network retries.
</ParamField>

### Body

<ParamField body="amount" type="number" required>
  Amount in major currency units (e.g. `25000` for 25,000 NGN).
</ParamField>
<ParamField body="sourceCurrency" type="string" required>
  Source wallet currency debited (e.g. `NGN`, `USD`).
</ParamField>
<ParamField body="payoutCurrency" type="string" required>
  Destination currency received by the beneficiary (e.g. `NGN`, `USD`).
</ParamField>
<ParamField body="recipientName" type="string" required>
  Full legal name of the beneficiary.
</ParamField>
<ParamField body="bankCode" type="string">
  3-digit CBN bank code (for Nigerian NIP payouts).
</ParamField>
<ParamField body="accountNumber" type="string">
  10-digit NUBAN account number (for Nigerian NIP payouts).
</ParamField>
<ParamField body="routing" type="string">
  9-digit ABA routing number (for US ACH/Wire payouts).
</ParamField>
<ParamField body="iban" type="string">
  International Bank Account Number (for EUR SEPA payouts).
</ParamField>
<ParamField body="sortCode" type="string">
  6-digit UK sort code (for GBP Faster Payments).
</ParamField>
<ParamField body="quoteId" type="string">
  Guaranteed quote ID from `POST /v1/fx/quotes` if performing a cross-currency payout.
</ParamField>
<ParamField body="narration" type="string">
  Transfer memo or invoice reference displayed on the recipient's bank statement.
</ParamField>

### Response

<ResponseField name="id" type="string">
  Unique payout identifier.
</ResponseField>
<ResponseField name="status" type="string">
  Current status: `PROCESSING`, `SUCCESS`, or `FAILED`.
</ResponseField>
<ResponseField name="reference" type="string">
  Unique financial transaction reference.
</ResponseField>
<ResponseField name="amount" type="number">
  Amount dispatched.
</ResponseField>
<ResponseField name="currency" type="string">
  Destination currency.
</ResponseField>
<ResponseField name="recipientName" type="string">
  Beneficiary name.
</ResponseField>
<ResponseField name="createdAt" type="string">
  ISO timestamp when payout was initiated.
</ResponseField>

```javascript Node.js
import { randomUUID } from "node:crypto";

const response = await fetch("https://api.tabsglobal.co/v1/payouts", {
  method: "POST",
  headers: {
    "Authorization": `Bearer ${process.env.TABS_SECRET_KEY}`,
    "Idempotency-Key": randomUUID(),
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    amount: 25000,
    sourceCurrency: "NGN",
    payoutCurrency: "NGN",
    bankCode: "058",
    accountNumber: "0123456789",
    recipientName: "ADEWALE ADELEKE",
    narration: "Invoice settlement #1042",
  }),
});

const payout = await response.json();
console.log(payout);
```

```python Python
import uuid
import requests

url = "https://api.tabsglobal.co/v1/payouts"
headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Idempotency-Key": str(uuid.uuid4()),
    "Content-Type": "application/json",
}
payload = {
    "amount": 25000,
    "sourceCurrency": "NGN",
    "payoutCurrency": "NGN",
    "bankCode": "058",
    "accountNumber": "0123456789",
    "recipientName": "ADEWALE ADELEKE",
    "narration": "Invoice settlement #1042",
}

response = requests.post(url, headers=headers, json=payload)
print(response.json())
```
